6 sqlite3

import sqlite3
con = sqlite3.connect(":memory:")
con.execute("CREATE TABLE teile (name TEXT, masse REAL)")
con.executemany("INSERT INTO teile VALUES (?, ?)", [("Welle", 2.4), ("Flansch", 5.1), ("Ring", 0.8)])
for zeile in con.execute("SELECT name, masse FROM teile WHERE masse > ? ORDER BY masse", (1.0,)):
    print(zeile)
con.close()
